  J: '#8b4513',

};


function drawPixelArt(canvas, grid, scale=4) {

  const rows = grid.length, cols = grid[0].length;

  canvas.width = cols; canvas.height = rows;

  canvas.style.width = cols*scale+'px'; canvas.style.height = rows*scale+'px';

  const ctx = canvas.getContext('2d');

  ctx.imageSmoothingEnabled = false;

  grid.forEach((row,y) => row.split('').forEach((c,x) => {

    const col = PALETTE[c];

    if (col) { ctx.fillStyle = col; ctx.fillRect(x,y,1,1); }

  }));

}

// simple check: does cart contain keywords from daily order?

function checkDailyOrder() {

  const daily = DAILY_ORDERS[dailyOrderIdx];

  const cartNames = cart.map(i => i.name + (i.isSet ? ' 세트' : ''));

  if (daily.includes('불고기버거 세트') && !cart.some(i => i.name === '불고기버거' && i.isSet)) return false;

  if (daily.includes('불갈비버거 세트') && !cart.some(i => i.name === '불갈비버거' && i.isSet)) return false;

  if (daily.includes('치즈버거') && !daily.includes('세트') && !cart.some(i => i.name === '치즈버거')) return false;

  return true; // lenient: pass if burger condition met

}

}

  pendingItem = null;

  document.getElementById('set-overlay').classList.add('hidden');

}

function setNo() {

  if (!pendingItem) return;

  pushToCart(pendingItem.name, pendingItem.price, false);

  pendingItem = null;

  document.getElementById('set-overlay').classList.add('hidden');

}


function pushToCart(name, price, isSet) {

  const label = name + (isSet ? ' 세트' : '');

  const existing = cart.find(i => i.label === label);

  if (existing) { existing.qty++; existing.totalPrice += price; }

  else cart.push({label, name, price, isSet, qty:1, totalPrice:price});

  renderCart();

}


function removeFromCart(label) {

  cart = cart.filter(i => i.label !== label);

  renderCart();

}


function renderCart() {

  const list = document.getElementById('cart-list');

  const count = document.getElementById('cart-count');

  const total = document.getElementById('total-price');

  const payBtn = document.getElementById('pay-btn');

  const t = getTotal();

  total.textContent = formatPrice(t);

  count.textContent = cart.reduce((s,i)=>s+i.qty,0);

  payBtn.disabled = cart.length === 0;

  if (cart.length === 0) {

    list.innerHTML = '<div class="cart-empty">담긴 메뉴가 없습니다</div>';

    return;

  }

  list.innerHTML = '';

  cart.forEach(item => {

    const row = document.createElement('div');

    row.className = 'cart-item';

    row.innerHTML = `

      <span class="cart-item-name">${item.label}</span>

      <span class="cart-item-qty">x${item.qty}</span>

      <span class="cart-item-price">${formatPrice(item.totalPrice)}원</span>

      <button class="cart-item-del px-btn" onclick="removeFromCart('${item.label}')">삭제</button>

    `;

    list.appendChild(row);

  });

}


function getTotal() {

  return cart.reduce((s,i) => s + i.totalPrice, 0);

}
